# COMANDOS DE MIGRACIÓN PARA HEROKU
# Ejecutar después de hacer push de los cambios

# 1. Verificar el estado actual de migraciones en Heroku
heroku run python manage.py showmigrations --app mi-gestion-servicio

# 2. Aplicar todas las migraciones pendientes (incluye la limpieza de visor_dicom)
heroku run python manage.py migrate --app mi-gestion-servicio

# 3. Verificar que las migraciones se aplicaron correctamente
heroku run python manage.py showmigrations --app mi-gestion-servicio

# 4. Opcional: Verificar que la tabla visor_dicom_dicomfile fue eliminada
heroku run python manage.py dbshell --app mi-gestion-servicio
# Dentro de la consola de base de datos ejecutar:
# \dt (en PostgreSQL) o .tables (en SQLite) para ver las tablas disponibles

# 5. Hacer push de los cambios al repositorio antes de migrar
git add .
git commit -m "feat: eliminar completamente visor_dicom - limpiar código, templates, archivos y DB"
git push heroku feature/tailwind-css-integration:main
